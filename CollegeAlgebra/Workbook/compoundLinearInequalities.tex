\begin{problem}
Which of the following is the solution to $-1 < 2x + 3 \leq 7$?
\begin{hint}
Subtract 3 from all parts, then divide by 2.
\end{hint}

\choice[correct]{$(-2, 2]$}
\choice{$[-2, 2)$}
\choice{$(-\infty, 2]$}
\end{multipleChoice}
\end{problem}

\begin{problem}
Solve $|2x - 1| \leq 5$ and express your answer in interval notation.
\begin{freeResponse}
Write your solution here.
\end{freeResponse}
\begin{solution}
Isolate: $-5 \leq 2x - 1 \leq 5$\\
Add 1: $-4 \leq 2x \leq 6$\\
Divide: $-2 \leq x \leq 3$\\
Answer: $[-2, 3]$
\end{solution}
\end{problem}
