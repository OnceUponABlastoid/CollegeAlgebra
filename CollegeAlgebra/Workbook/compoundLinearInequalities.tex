\section*{Quick Check}

\begin{problem}
Which of the following expressions has infinitely many solutions?
\begin{hint}
Simplify each expression. Look for one where the variable cancels and the resulting statement is always true.
\end{hint}
\begin{multipleChoice}
  \choice{$2x+3 > 2x+5$}
  \choice[correct]{$3(x-1) < 3x+2$}
  \choice{$2(x+4) > 2x+9$}
  \choice{$x+3 < x-1$}
\end{multipleChoice}
\end{problem}

\begin{problem}
The inequality $3(x-2) < 3x - 6$ is:
\begin{hint}
Distribute the left side and simplify. What does the resulting statement say?
\end{hint}
\begin{multipleChoice}
  \choice{Always true}
  \choice[correct]{Always false}
  \choice{True only for $x > 0$}
  \choice{True only for $x < 2$}
\end{multipleChoice}
\end{problem}

\begin{problem}
Which of the following statements is always true?
\begin{hint}
Consider what happens when you multiply both sides of $3 < a$ by a positive number $x$.
\end{hint}
\begin{multipleChoice}
  \choice{If $3x < ax$, then $3 < a$.}
  \choice{If $3 < a$, then $3x < ax$ for all real numbers $x$.}
  \choice[correct]{If $3 < a$, then $3x < ax$ for all \textit{positive} real numbers $x$.}
  \choice{If $3x < ax$, then $3 = a$.}
\end{multipleChoice}
\end{problem}

\begin{problem}
The compound inequality $4 < 2x + 1 < 10$ is logically equivalent to:
\begin{hint}
Subtract 1 from all three parts, then divide by 2. Check whether endpoints should be included.
\end{hint}
\begin{multipleChoice}
  \choice{$x > \dfrac{3}{2}$ or $x < \dfrac{9}{2}$}
  \choice[correct]{$\dfrac{3}{2} < x < \dfrac{9}{2}$}
  \choice{$x < \dfrac{3}{2}$ or $x > \dfrac{9}{2}$}
  \choice{$\dfrac{3}{2} \leq x \leq \dfrac{9}{2}$}
\end{multipleChoice}
\end{problem}

\begin{problem}
Solve the compound inequality $-3 < 2 - x \leq 5$.
\begin{hint}
Subtract 2 from all three parts, then multiply by $-1$ — remember to flip both inequality signs.
\end{hint}
\begin{multipleChoice}
  \choice{$(-3,5)$}
  \choice[correct]{$[-3,5)$}
  \choice{$(-3,5]$}
  \choice{$[-5,3)$}
\end{multipleChoice}
\end{problem}

\begin{problem}
Solve the following inequalities. Express your solutions in (i) set notation and (ii) interval
notation, and (iii) graph the solution set.
\begin{enumerate}
  \item[(a)] $\ds 5 \leq -\dfrac{3}{4}t + 10$
  \medskip
  \item[(b)] $2(3x+1) - 4x > 2(x+8) - 5$
  \medskip
  \item[(c)] $\ds\frac{4+x}{2} - \frac{x-3}{5} < -\frac{x}{10}$
  \medskip
  \item[(d)] $8 - 4\big[4 - 3(y-3)\big] \geq 5\Big\{4-2\big[5-(y-3)\big]\Big\}$
\end{enumerate}

\begin{expandable}
\begin{enumerate}
  \item[(a)] Isolate $t$.
  \begin{align*}
    -5 &\leq -\frac{3}{4}t \\
    \frac{20}{3} &\geq t
  \end{align*}
  (i) Set notation: $\ds\left\{t\;\middle|\;t\leq \frac{20}{3}\right\}$\\
  (ii) Interval notation: $\ds\left(-\infty,\frac{20}{3}\right]$

  \item[(b)] Distribute the 2's and isolate $x$.
  \begin{align*}
    6x + 2 - 4x &> 2x + 16 - 5\\
    2x+2 &> 2x + 11\\
    2 &> 11
  \end{align*}
  Since the last inequality is clearly false, this means no $x$ can ever satisfy the original
  inequality, so it has \textbf{NO SOLUTION}.

  \item[(c)] Multiply all terms by LCD (10) to clear the fractions; distribute and isolate $x$.
  \begin{align*}
    10\!\left(\frac{4+x}{2}\right) - 10\!\left(\frac{x-3}{5}\right) &< -10\!\left(\frac{x}{10}\right)\\
    5(4+x) - 2(x-3) &< -x\\
    20 + 5x - 2x + 6 &< -x\\
    26 + 3x &< -x\\
    26 &< -4x\\
    -\frac{13}{2} &> x
  \end{align*}
  (i) Set notation: $\ds\left\{x\;\middle|\;x< -\frac{13}{2}\right\}$\\
  (ii) Interval notation: $\ds\left(-\infty,-\frac{13}{2}\right)$

  \item[(d)] Distribute (starting from the inside parentheses first) and isolate $y$.
  \begin{align*}
    8 - 4\big[4 - 3y+9\big] &\geq 5\Big\{4-2\big[5-y+3\big]\Big\}\\
    8 - 16 + 12y - 36 &\geq 5\Big\{4 - 10 + 2y - 6\Big\}\\
    8 - 16 + 12y - 36 &\geq 20 - 50 + 10y - 30\\
    -44 + 12y &\geq -60 + 10y\\
    2y &\geq -16\\
    y &\geq -8
  \end{align*}
  (i) Set notation: $\{y \mid y\geq -8\}$\\
  (ii) Interval notation: $[-8,\infty)$
\end{enumerate}
\end{expandable}
\end{problem}


\begin{problem}
Solve the following compound inequalities. Express your solutions in (i) interval notation
and (ii) graph the solution set.
\begin{enumerate}
  \item[(a)] $z - 6 > 1 \AND \dfrac{3}{4}z \geq 6$
  \medskip
  \item[(b)] $z - 6 > 1 \OR \dfrac{3}{4}z \geq 6$
  \medskip
  \item[(c)] $-6 \leq -3x+9 < 0$
  \medskip
  \item[(d)] $\ds -2 \leq \frac{-2x+1}{-3} < 4$
\end{enumerate}

\begin{expandable}
\begin{enumerate}
  \item[(a)] Solving for $z$ in each inequality, we get
  $$z > 7 \AND z \geq 8.$$
  We are looking for the intersection (overlap) of the two solution sets: $$[8,\infty)$$
  \textit{Hint: Draw the two solutions on separate number lines and see where they overlap.}

  \begin{tikzpicture}[scale=0.45]
    \draw[<->,thick] (4.5,0)--(12.5,0);
    \foreach \x in {5,6,7,8,9,10,11,12} {
      \draw[thick] (\x,0.2)--(\x,-0.2) node[below,font=\small] {$\x$};
    }
    \draw[->,very thick,blue] (8,0)--(12.5,0);
    \closeddot{8}
  \end{tikzpicture}

  \item[(b)] The inequalities are the same as in part (a). Now we are looking for the union
  (addition) of the two solution sets:
  $$(7,\infty)$$

  \begin{tikzpicture}[scale=0.45]
    \draw[<->,thick] (4.5,0)--(12.5,0);

  \item[(c)] Isolate $x$ in the middle ``section."
  \begin{align*}
    -15 &\leq -3x < -9\\

  Interval notation: $(3,5]$

  \item[(d)] Isolate $x$ in the middle ``section."
  \begin{align*}
    6 &\geq -2x+1 > -12\\
    5 &\geq -2x > -13\\

\end{enumerate}

\begin{problem}
Ann earns scores of 78, 82, 90, 80, and 75 on her five chapter tests for her nursing class
and a grade of 85 on the class project. The overall average for the course is computed as follows:
\begin{itemize}
  \item The average of the five chapter tests makes up 60\% of the course grade
  \item The project accounts for 10\% of the grade
  \item The final exam accounts for 30\% of the grade
\end{itemize}
What scores can Ann earn on the final exam to earn a ``B'' in the course if the cut-off for a
``B'' is an overall score greater than or equal to 80, but less than 90? \textit{Assume that 100
is the highest score that can be earned on the final exam and that only whole-number scores
are given.}
\begin{hint}
Let $x$ = Ann's score on the final. Write a compound inequality for the overall grade and solve for $x$.
\end{hint}
\begin{expandable}
Let $x = $ Ann's score on the final. Based on the grade cut-off, Ann needs
\begin{align*}
  80 &\leq \text{Course grade} < 90\\
  80 &\leq 0.6\!\left(\frac{78+82+90+80+75}{5}\right) + 0.1(85) + 0.3x < 90\\
  80 &\leq 0.6(81) + 0.1(85) + 0.3x < 90\\
  80 &\leq 48.6 + 8.5 + 0.3x < 90\\
  80 &\leq 57.1 + 0.3x < 90\\
  22.9 &\leq 0.3x < 32.9\\
  76\tfrac{1}{3} &\leq x < 109\tfrac{2}{3}
\end{align*}
Since 100 is the highest score that can be earned on the final and only whole-number scores
are given, \textbf{Ann needs to earn a final exam score between 77 and 100, inclusive, in order
to earn a ``B'' in the course.}
\end{expandable}
\end{problem}


\begin{problem}
The lengths of the sides of a triangle are given by three consecutive integers greater than 1.
What are the possible values for the shortest side if the perimeter is not to exceed 24 ft?
\begin{hint}
Let $x$ = the length of the shortest side. The other sides are $x+1$ and $x+2$.
\end{hint}
\begin{expandable}
Let $x = $ the length of the shortest side of the triangle. Then the other sides will be
given by $x+1$ and $x+2$. We want
\begin{align*}
  \text{Perimeter} &\leq 24\\
  x + (x+1) + (x+2) &\leq 24\\
  3x + 3 &\leq 24\\
  3x &\leq 21\\
  x &\leq 7
\end{align*}
Since we require that each side is greater than 1, \textbf{the possible values for the shortest
side are 2 ft, 3 ft, 4 ft, 5 ft, 6 ft, and 7 ft.}
\end{expandable}
\end{problem}


\begin{problem}
Tom has offers from two sales jobs. Job A has a base salary of \$25{,}000 and pays a 10\%
commission on sales. Job B has a base salary of \$35{,}000 and pays an 8\% commission on
sales. How much would Tom have to sell for the salary from Job A to exceed the salary from Job B?
\begin{hint}
Let $x$ = the amount of sales (in \$). Total salary = base salary + commission on sales.
\end{hint}
\begin{expandable}
The unknown is the \textbf{amount of sales}, which we can call $x$ (in \$). Total salary has
two components:
\begin{enumerate}
  \item Base salary: given amount (\$25{,}000 or \$35{,}000 depending on the job)
  \item Commission on sales: \% of sales amount $x$ (either $0.1x$ or $0.08x$ depending on the job)
\end{enumerate}
Based on this information,
\begin{align*}
  \text{Total salary from Job A} &= 25000 + 0.1x\\
  \text{Total salary from Job B} &= 35000 + 0.08x
\end{align*}
So our desired inequality to solve is
\begin{align*}
  25000 + 0.1x &> 35000 + 0.08x\\
  0.1x &> 10000 + 0.08x\\
  0.02x &> 10000\\
  x &> 500{,}000
\end{align*}
\textbf{Tom would have to sell more than \$500{,}000 of product for the total salary from
Job A to exceed the total salary from Job B.}
\end{expandable}
\end{problem}
