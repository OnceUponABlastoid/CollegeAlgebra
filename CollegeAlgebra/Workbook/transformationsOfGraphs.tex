\section*{Quick Check}

\begin{problem}
If the domain of $f(x)$ is $[-3,1]$, what is the domain of $g(x) = 2f(-x+1)$?
\begin{hint}
Solve the inequality $-3 \leq -x+1 \leq 1$ to find which $x$-values keep the argument of $f$ within its domain.
\end{hint}
\begin{multipleChoice}
  \choice{$[-2,2]$}
  \choice{$[-6,2]$}
  \choice{$[-1,3]$}
  \choice[correct]{$[0,4]$}
\end{multipleChoice}
\end{problem}

\begin{problem}
The function $g(x)$ passes through $(4,-1)$ and has a vertical asymptote at $x=1$. What can you say about $h(x) = 3g(-x-2)+1$?
\begin{hint}
To find where a known point $(a,b)$ on $g$ maps to on $h$: solve $-x-2 = a$ for the new $x$-coordinate, then compute $3b+1$ for the new $y$-coordinate.
\end{hint}
\begin{multipleChoice}
  \choice{Must pass through $(13,3)$.}
  \choice{Must have a vertical asymptote at $x=4$.}
  \choice{Must have a vertical asymptote at $y=-3$.}
  \choice[correct]{Must pass through $(-6,-2)$.}
\end{multipleChoice}
\end{problem}

\begin{problem}
The following transformations are applied in order to $f(x) = \sqrt{x}$: (1) shift left 3, (2) shift up 1, (3) reflect over the $y$-axis. Which is the resulting function $g(x)$?
\begin{hint}
Apply each transformation one at a time to the function expression.
\end{hint}
\begin{multipleChoice}
  \choice[correct]{$g(x) = \sqrt{-x+3}+1$}
  \choice{$g(x) = -\sqrt{x+3}-1$}
  \choice{$g(x) = -\sqrt{x+3}+1$}
  \choice{$g(x) = \sqrt{-x-3}+1$}
\end{multipleChoice}
\end{problem}

\begin{problem}
The graph of $y = f(x)$ passes through $(2,5)$. Which of the following must be a point on the graph of $y = f(x+3) - 2$?
\begin{hint}
If $(a,b)$ is on $f$, then $(a-3,\,b-2)$ is on $f(x+3)-2$.
\end{hint}
\begin{multipleChoice}
  \choice[correct]{$(-1,3)$}
  \choice{$(5,3)$}
  \choice{$(-1,7)$}
  \choice{$(5,7)$}
\end{multipleChoice}
\end{problem}

\section*{Extended Practice}

\begin{problem}
Given $f(x)$, describe the transformation(s) needed to obtain each function.
\begin{enumerate}
  \item[(a)] $f(x+3)$
  \medskip
  \item[(b)] $2 - f(x)$
  \medskip
  \item[(c)] $f(3x)$
  \medskip
  \item[(d)] $f(-x-6)$
  \medskip
  \item[(e)] $f(x) + 3$
  \medskip
  \item[(f)] $3f(x)$
\end{enumerate}

\begin{solution}
\begin{enumerate}
  \item[(a)] Shift \textbf{left} 3 units.

  \item[(b)] Rewrite as $-f(x)+2$: (i) reflect over the $x$-axis, (ii) shift up 2 units.

  \item[(c)] Shrink \textbf{horizontally} by a factor of $\dfrac{1}{3}$.

  \item[(d)] Rewrite as $f(-(x+6))$: (i) shift right 6 units, (ii) reflect over the $y$-axis.

  \item[(e)] Shift \textbf{up} 3 units.

  \item[(f)] Stretch \textbf{vertically} by a factor of 3.
\end{enumerate}
\end{solution}
\end{problem}


\begin{problem}
For each function, (i) identify the parent function, (ii) list the transformations in order.
\begin{enumerate}
  \item[(a)] $m(x) = -\left|\dfrac{1}{3}x+1\right|$
  \medskip
  \item[(b)] $\ds g(x) = -\dfrac{1}{3}(x+2)^2 + 3$
\end{enumerate}

\begin{solution}
\begin{enumerate}
  \item[(a)] Parent: $f(x) = |x|$.\\
  Transformations: (i) shift left 1 unit, (ii) stretch horizontally by a factor of 3, (iii) reflect over the $x$-axis.

  \item[(b)] Parent: $f(x) = x^2$.\\
  Transformations: (i) shift left 2 units, (ii) shrink vertically by a factor of $\dfrac{1}{3}$, (iii) reflect over the $x$-axis, (iv) shift up 3 units.
\end{enumerate}
\end{solution}
\end{problem}


\begin{problem}
Given the parent function, apply the transformations in order to find $g(x)$.
\begin{enumerate}
  \item[(a)] $f(x) = \sqrt[3]{x}$: (i) shift 1 unit left, (ii) stretch horizontally by a factor of 4, (iii) reflect over the $x$-axis.
  \medskip
  \item[(b)] $f(x) = \dfrac{1}{x}$: (i) shift 2 units left, (ii) reflect over the $y$-axis, (iii) stretch vertically by a factor of 3, (iv) shift down 1 unit.
\end{enumerate}

\begin{solution}
\begin{enumerate}
  \item[(a)] Step by step:
  \begin{itemize}
    \item Shift left 1: $\sqrt[3]{x+1}$
    \item Stretch horizontally by 4: $\sqrt[3]{\dfrac{x}{4}+1}$
    \item Reflect over $x$-axis: $g(x) = -\sqrt[3]{\dfrac{x}{4}+1}$
  \end{itemize}

  \item[(b)] Step by step:
  \begin{itemize}
    \item Shift left 2: $\dfrac{1}{x+2}$
    \item Reflect over $y$-axis: $\dfrac{1}{-x+2}$
    \item Stretch vertically by 3: $\dfrac{3}{-x+2}$
    \item Shift down 1: $g(x) = \dfrac{3}{-x+2} - 1$
  \end{itemize}
\end{enumerate}
\end{solution}
\end{problem}


\begin{problem}
Describe the transformations that map $f(x)$ to each indicated function.
\begin{enumerate}
  \item[(a)] $y = -f(x-1) + 2$
  \medskip
  \item[(b)] $\ds y = -\dfrac{1}{2}f\!\left(\dfrac{1}{2}x\right)$
\end{enumerate}

\begin{solution}
\begin{enumerate}
  \item[(a)] (i) Shift right 1 unit. (ii) Reflect over the $x$-axis. (iii) Shift up 2 units.

  \item[(b)] (i) Stretch horizontally by a factor of 2. (ii) Shrink vertically by a factor of $\dfrac{1}{2}$. (iii) Reflect over the $x$-axis.
\end{enumerate}
\end{solution}
\end{problem}


\begin{problem}
Using transformations on basic functions, write a function $y = f(x)$ for each description below.
\begin{enumerate}
  \item[(a)] The graph is a square root function shifted left 3, stretched vertically by 2, and shifted down 1.
  \medskip
  \item[(b)] The graph is a reciprocal function ($\frac{1}{x}$) shifted right 4, stretched vertically by 5, and shifted up 1.
\end{enumerate}

\begin{solution}
\begin{enumerate}
  \item[(a)] $f(x) = 2\sqrt{x+3} - 1$

  \item[(b)] $f(x) = \dfrac{5}{x-4} + 1$
\end{enumerate}
\end{solution}
\end{problem}
