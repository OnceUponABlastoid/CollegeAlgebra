\begin{problem}
Given $g(x) = \sqrt{2x+3}$, on which of the following intervals is the average rate of change the greatest?
\begin{hint}
The rate of change of a concave-down function is greatest at the left end of its domain. Check which intervals are in the domain ($2x+3 \geq 0$).
\end{hint}
\begin{multipleChoice}
  \choice{$[1,2]$}
  \choice{$[-3,-2]$}
  \choice{$[0,1]$}
  \choice{$[-2,-1]$}
  \choice[correct]{$[-1,0]$}
\end{multipleChoice}
\end{problem}

\begin{problem}
What is the slope of the line passing through $(-3,4)$ and parallel to the $y$-axis?
\begin{hint}
A line parallel to the $y$-axis is a vertical line.
\end{hint}
\begin{multipleChoice}
  \choice{$0$}
  \choice{$\infty$}
  \choice{$-\dfrac{4}{3}$}
  \choice[correct]{Undefined}
\end{multipleChoice}
\end{problem}

\begin{problem}
Which of the following is the equation of the line perpendicular to $y = \dfrac{1}{3}x - 5$ and passing through $(1,2)$?
\begin{hint}
The slope of a perpendicular line is the negative reciprocal of the original slope.
\end{hint}
\begin{multipleChoice}
  \choice[correct]{$y = -3x + 5$}
  \choice{$y = 3x - 1$}
  \choice{$y = -3x - 5$}
  \choice{$y = \dfrac{1}{3}x + 2$}
\end{multipleChoice}
\end{problem}

\begin{problem}
A driver leaves a rest stop and travels at a constant speed. After 2 hours she is 130 miles from home; after 5 hours she is 280 miles from home. What is her average speed in miles per hour?
\begin{hint}
Average speed is the rate of change: $\dfrac{\Delta \text{distance}}{\Delta \text{time}}$.
\end{hint}
\begin{multipleChoice}
  \choice{40 mph}
  \choice{45 mph}
  \choice[correct]{50 mph}
  \choice{56 mph}
\end{multipleChoice}
\end{problem}

\section*{Extended Practice}

\begin{problem}
Determine the slope of the line passing through the given points.
\begin{enumerate}
  \item[(a)] $(17,-9)$ and $(42,-6)$
  \medskip
  \item[(b)] $\left(\dfrac{3}{4},6\right)$ and $\left(\dfrac{5}{2},1\right)$
\end{enumerate}

\begin{solution}
\begin{enumerate}
  \item[(a)] $\ds m = \frac{-9-(-6)}{17-42} = \frac{-3}{-25} = \frac{3}{25}$

  \item[(b)] $\ds m = \frac{6-1}{\frac{3}{4}-\frac{5}{2}} = \frac{5}{\frac{3}{4}-\frac{10}{4}} = \frac{5}{-\frac{7}{4}} = -\frac{20}{7}$
\end{enumerate}
\end{solution}
\end{problem}


\begin{problem}
For each equation, (i) rewrite in slope-intercept form and (ii) identify the slope and $y$-intercept.
\begin{enumerate}
  \item[(a)] $2x - 4y = 8$
  \medskip
  \item[(b)] $\ds\frac{x}{4} + \frac{y}{7} = 1$
\end{enumerate}

\begin{solution}
\begin{enumerate}
  \item[(a)] Solve for $y$:
  \begin{align*}
    -4y &= -2x + 8\\
    y &= \frac{1}{2}x - 2
  \end{align*}
  Slope $= \dfrac{1}{2}$; $y$-intercept: $(0,-2)$

  \item[(b)] Solve for $y$:
  \begin{align*}
    \frac{y}{7} &= -\frac{x}{4} + 1\\
    y &= -\frac{7}{4}x + 7
  \end{align*}
  Slope $= -\dfrac{7}{4}$; $y$-intercept: $(0,7)$
\end{enumerate}
\end{solution}
\end{problem}


\begin{problem}
Use slope-intercept form to write the equation of the line that passes through the given point and has the given slope.
\begin{enumerate}
  \item[(a)] $(0,-4)$; $m = \dfrac{1}{3}$
  \medskip
  \item[(b)] $(-4,-2)$; $m = \dfrac{3}{2}$
\end{enumerate}

\begin{solution}
\begin{enumerate}
  \item[(a)] The given point is the $y$-intercept ($b=-4$):
  $$y = \frac{1}{3}x - 4$$

  \item[(b)] Substitute $y=mx+b$ with the given point to find $b$:
  \begin{align*}
    -2 &= \frac{3}{2}(-4) + b\\
    -2 &= -6 + b\\
    b &= 4
  \end{align*}
  Equation: $y = \dfrac{3}{2}x + 4$
\end{enumerate}
\end{solution}
\end{problem}


\begin{problem}
Write the equation of the line that passes through $(-3,5)$ and has an undefined slope.

\begin{solution}
A line with an \textbf{undefined} slope is a \textbf{vertical} line: $x = -3$.
\end{solution}
\end{problem}


\begin{problem}
Write the equation of the line that passes through $(2,1)$ and has a slope of $0$.

\begin{solution}
A line with \textbf{slope 0} is a \textbf{horizontal} line: $y = 1$.
\end{solution}
\end{problem}


\begin{problem}
Use slope-intercept form to write the equation of the line through the given points.
\begin{enumerate}
  \item[(a)] $(-8,1)$ and $(0,-3)$
  \medskip
  \item[(b)] $(7,-3)$ and $(4,1)$
\end{enumerate}

\begin{solution}
\begin{enumerate}
  \item[(a)] Slope: $m = \dfrac{-3-1}{0-(-8)} = -\dfrac{1}{2}$. The second point $(0,-3)$ is the $y$-intercept, so $b=-3$:
  $$y = -\frac{1}{2}x - 3$$

  \item[(b)] Slope: $m = \dfrac{-3-1}{7-4} = -\dfrac{4}{3}$. Using point $(4,1)$:
  \begin{align*}
    1 &= -\frac{4}{3}(4) + b\\
    1 &= -\frac{16}{3} + b\\
    b &= \frac{19}{3}
  \end{align*}
  Equation: $y = -\dfrac{4}{3}x + \dfrac{19}{3}$
\end{enumerate}
\end{solution}
\end{problem}


\begin{problem}
Determine the average rate of change of the function on the given interval.
\begin{enumerate}
  \item[(a)] $f(x) = x^2 - 3$; $[-2,0]$
  \medskip
  \item[(b)] $g(x) = \sqrt{x-1}$; $[5,10]$
\end{enumerate}

\begin{solution}
\begin{enumerate}
  \item[(a)] \begin{align*}
    \frac{f(0)-f(-2)}{0-(-2)} &= \frac{[(0)^2-3]-[(-2)^2-3]}{2} = \frac{-3-1}{2} = -2
  \end{align*}

  \item[(b)] \begin{align*}
    \frac{g(10)-g(5)}{10-5} &= \frac{\sqrt{9}-\sqrt{4}}{5} = \frac{3-2}{5} = \frac{1}{5}
  \end{align*}
\end{enumerate}
\end{solution}
\end{problem}


\begin{problem}
Find the $x$- and $y$-intercepts of each line.
\begin{enumerate}
  \item[(a)] $-3x + 4y = 12$
  \medskip
  \item[(b)] $0.02x + 0.05y = 0.1$
\end{enumerate}

\begin{solution}
\begin{enumerate}
  \item[(a)] $x$-intercept ($y=0$): $-3x=12 \Rightarrow x=-4$. Point: $(-4,0)$.\\
  $y$-intercept ($x=0$): $4y=12 \Rightarrow y=3$. Point: $(0,3)$.

  \item[(b)] $x$-intercept ($y=0$): $0.02x=0.1 \Rightarrow x=5$. Point: $(5,0)$.\\
  $y$-intercept ($x=0$): $0.05y=0.1 \Rightarrow y=2$. Point: $(0,2)$.
\end{enumerate}
\end{solution}
\end{problem}


\begin{problem}
\textbf{Challenge.} Determine the area enclosed by the three lines:
$$y = x+6, \quad y = -2x+6, \quad y = 0$$

\begin{solution}
Find the three vertices by solving pairs of equations:
\begin{itemize}
  \item $y=x+6$ and $y=0$: $x=-6$, giving $(-6,0)$
  \item $y=-2x+6$ and $y=0$: $x=3$, giving $(3,0)$
  \item $y=x+6$ and $y=-2x+6$: $x+6=-2x+6 \Rightarrow x=0$, $y=6$, giving $(0,6)$
\end{itemize}
The three lines form a triangle with base $= 3-(-6) = 9$ and height $= 6$:
$$\text{Area} = \frac{1}{2}(9)(6) = 27$$
\end{solution}
\end{problem}
